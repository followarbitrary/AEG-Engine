cmake_minimum_required(VERSION 3.25)
project(AEGEngine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(cmake/ProjectOptions.cmake)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Vulkan 1.2 REQUIRED COMPONENTS glslc)
include(cmake/Dependencies.cmake)
include(cmake/VulkanShaders.cmake)
include(CTest)
option(AEG_ENABLE_GPU_TESTS "Run GPU smoke tests (requires a display and Vulkan device)" OFF)

add_subdirectory(engine/foundation)
add_subdirectory(engine/camera)
add_subdirectory(engine/assets)
add_subdirectory(engine/rhi)
add_subdirectory(engine/gui)
add_subdirectory(engine/rhi/vulkan)
add_subdirectory(engine/render)
aeg_compile_vulkan_shaders(aeg_shaders "${PROJECT_SOURCE_DIR}/engine/render/private/shaders" aeg_shader_headers)
target_sources(aeg_render PRIVATE ${aeg_shader_headers})
target_include_directories(aeg_render PRIVATE "${CMAKE_BINARY_DIR}/generated")
add_dependencies(aeg_render aeg_shaders)
add_subdirectory(editor)
add_subdirectory(apps/editor)

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
