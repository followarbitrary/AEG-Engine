foreach(backend IN ITEMS normal scalar)
    add_executable(aeg_math_tests_${backend} MathTests.cpp)
    target_compile_features(aeg_math_tests_${backend} PRIVATE cxx_std_23)
    if(backend STREQUAL "scalar")
        target_link_libraries(aeg_math_tests_${backend} PRIVATE aeg::math_scalar)
    else()
        target_link_libraries(aeg_math_tests_${backend} PRIVATE aeg::math)
    endif()
    add_test(NAME aeg_math_tests_${backend} COMMAND aeg_math_tests_${backend})
    set_tests_properties(aeg_math_tests_${backend} PROPERTIES LABELS "cpu;math")
    if(MINGW)
        get_filename_component(aeg_math_compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
        set_tests_properties(aeg_math_tests_${backend} PROPERTIES
            ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${aeg_math_compiler_directory}")
    endif()
endforeach()
add_library(aeg_math_public_header OBJECT MathHeaderTests.cpp)
target_link_libraries(aeg_math_public_header PRIVATE aeg::math)
