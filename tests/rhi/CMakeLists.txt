file(GLOB aeg_rhi_public_headers "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/rhi/public/*.hpp")
foreach(header IN LISTS aeg_rhi_public_headers)
    get_filename_component(header_name "${header}" NAME)
    get_filename_component(header_id "${header}" NAME_WE)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/header-${header_id}.cpp")
    file(WRITE "${source}" "#include <${header_name}>\n")
    add_library("aeg_rhi_header_${header_id}" OBJECT "${source}")
    target_link_libraries("aeg_rhi_header_${header_id}" PRIVATE aeg::rhi)
endforeach()
